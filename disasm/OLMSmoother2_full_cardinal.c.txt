// STRUCTURE OVERVIEW
// ==================
// 4 cardinal entry functions (called from main dispatch based on 2x2+neighbor key):
//   FUN_1800105f0 — "north-side"  (scans y+1 row, emits at y-1)
//   FUN_1800106b0 — "east-side"   (scans col, pattern TBD)
//   FUN_180010760 — "south-side"
//   FUN_180010820 — "west-side"
//
// Each entry sets up param_2 = 6-int scan descriptor:
//   param_2[0] = left/up scan x/y endpoint
//   param_2[1] = y (center)
//   param_2[2] = class_mask of left/up endpoint (from FUN_180010550)
//   param_2[3] = right/down scan x/y endpoint
//   param_2[4] = y (center, copy)
//   param_2[5] = class_mask of right/down endpoint
// Then calls emitter dispatcher FUN_18000fbf0/f8f0/fef0/1800101e0.
//
// DISPATCHER SWITCH KEY
// =====================
// FUN_18000fbf0/f8f0:  key = param_2[2] + (param_2[5]*5 - 1) * 2
// FUN_18000fef0:       key = param_2[2] + param_2[5]*10 - 1
// FUN_1800101e0:       key = (param_2[2] - 1) + param_2[5]*10
// All 4 share the SAME case layout (0, 6, 20, 26, 30, 36, 80, 86, 3, 8, 18, 43, 48, 53, 58, 78, etc.)
// The switch routes to 1-3 leaves per case via 8 leaf-families × 4 direction-rotations.
//
// JUMPTABLE TRUTH TABLE (FUN_180010550, RESOLVED)
// ================================================
// index = p1 + p3*2 + p2*4 where (p1, p2, p3) are 3 bool args:
//   (0,0,0) -> 0   (0,0,1) -> 1   (0,1,0) -> 2   (1,1,0) -> 3
//   (1,0,0) -> 0   (1,0,1) -> 4   (0,1,1) -> 5   (1,1,1) -> 6 (practically)
//
// EMIT FUNCTIONS (4 = one per perpendicular direction)
// =====================================================
// All four call FUN_180013630 trapezoid for weight then FUN_1800104d0 append:
//
//   FUN_18000e3a0(poly, p2, scale_m, scale_h):
//     weight = trap((p2[4] - p2[1] + 1) * scale_m * poly.smoothness_n,
//                   poly.cur_y - p2[1],
//                   scale_h * DAT_180022694);
//     emit at (p2[0], p2[1] - 1);
//     # Emits NORTH of scan origin row.
//
//   FUN_18000e290(poly, p2, scale_m, scale_h):
//     weight = trap((p2[4] - p2[1] + 1) * scale_m * poly.smoothness_n,
//                   p2[4] - poly.cur_y,
//                   scale_h * DAT_180022694);
//     emit at (p2[3], p2[4] + 1);
//     # Emits SOUTH of scan endpoint row.
//
//   FUN_18000e430(poly, p2, scale_m, scale_h):
//     weight = trap((p2[3] - p2[0] + 1) * scale_m * poly.smoothness_n,
//                   poly.cur_x - p2[0],
//                   scale_h * DAT_180022694);
//     emit at (p2[0] - 1, p2[1]);
//     # Emits WEST of scan origin col.
//
//   FUN_18000e320(poly, p2, scale_m, scale_h):
//     weight = trap((p2[3] - p2[0] + 1) * scale_m * poly.smoothness_n,
//                   p2[3] - poly.cur_x,
//                   scale_h * DAT_180022694);
//     emit at (p2[3] + 1, p2[4]);
//     # Emits EAST of scan endpoint col.
//
// DIRECTION-TO-DISPATCHER MAPPING
// ================================
// Cross-referenced via {scanner called, emit_fn used by leaves}:
//   FUN_1800105f0 (12時 entry) → FUN_18000fbf0 → leaves use e3a0 (NORTH emit)
//   FUN_1800106b0 (9時 entry?) → FUN_18000f8f0 → leaves use e3a0/e290 (N/S emit... oddly)
//   FUN_180010760 (6時 entry)  → FUN_18000fef0 → leaves use e3a0/e290
//   FUN_180010820 (3時 entry)  → FUN_1800101e0 → leaves use e430/e320 (W/E emit)
//
// This is the 4-way rotation: two dispatchers operate on horizontal runs (emit N/S),
// two on vertical runs (emit W/E).  Direction labels are the SCAN direction, not emit.
//
// LEAF MAPPING BY DIRECTION
